        let audioCtx = null;
        let isAudioMuted = false;

        function initAudio() {
            if (!audioCtx) {
                audioCtx = new (window.AudioContext || window.webkitAudioContext)();
            }
        }

        function playSound(type) {
            if (isAudioMuted || !audioCtx) return;
            try {
                let osc = audioCtx.createOscillator();
                let gain = audioCtx.createGain();
                osc.connect(gain);
                gain.connect(audioCtx.destination);

                if (type === 'chop') {
                    osc.type = 'triangle';
                    osc.frequency.setValueAtTime(120, audioCtx.currentTime);
                    osc.frequency.exponentialRampToValueAtTime(10, audioCtx.currentTime + 0.15);
                    gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
                    gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.15);
                    osc.start();
                    osc.stop(audioCtx.currentTime + 0.15);
                } else if (type === 'hit') {
                    osc.type = 'sawtooth';
                    osc.frequency.setValueAtTime(90, audioCtx.currentTime);
                    osc.frequency.exponentialRampToValueAtTime(10, audioCtx.currentTime + 0.2);
                    gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
                    gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.2);
                    osc.start();
                    osc.stop(audioCtx.currentTime + 0.2);
                } else if (type === 'craft') {
                    osc.type = 'sine';
                    osc.frequency.setValueAtTime(440, audioCtx.currentTime);
                    osc.frequency.setValueAtTime(880, audioCtx.currentTime + 0.1);
                    gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
                    gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.3);
                    osc.start();
                    osc.stop(audioCtx.currentTime + 0.3);
                } else if (type === 'freeze') {
                    osc.type = 'sawtooth';
                    osc.frequency.setValueAtTime(100, audioCtx.currentTime);
                    gain.gain.setValueAtTime(0.1, audioCtx.currentTime);
                    gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.5);
                    osc.start();
                    osc.stop(audioCtx.currentTime + 0.5);
                } else if (type === 'chime') {
                    osc.type = 'sine';
                    osc.frequency.setValueAtTime(523.25, audioCtx.currentTime); 
                    osc.frequency.setValueAtTime(659.25, audioCtx.currentTime + 0.12); 
                    osc.frequency.setValueAtTime(783.99, audioCtx.currentTime + 0.24); 
                    gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
                    gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.6);
                    osc.start();
                    osc.stop(audioCtx.currentTime + 0.6);
                }
            } catch(e){}
        }

        function playGunSound(type) {
            if (isAudioMuted || !audioCtx) return;
            try {
                let osc = audioCtx.createOscillator();

                    gain.gain.setValueAtTime(0.4, audioCtx.currentTime);
                    gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.35);
                    osc.start(); osc.stop(audioCtx.currentTime + 0.35);
                } else if (type === 'm4a1') {
                    osc.type = 'sawtooth';
                    osc.frequency.setValueAtTime(450, audioCtx.currentTime);
                    osc.frequency.exponentialRampToValueAtTime(15, audioCtx.currentTime + 0.15);

                    gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.15);
                    osc.start(); osc.stop(audioCtx.currentTime + 0.15);

            } catch(e){}

        }

        function logMsg(text, type = "info") {
            const box = document.getElementById('log-box');
            let color = "text-stone-300";
            if (type === "warning") color = "text-yellow-400 font-bold";
            if (type === "danger") color = "text-red-400 font-bold animate-pulse";
            if (type === "success") color = "text-green-400 font-bold";
            
            box.innerHTML += `<div class="${color}">[${new Date().toLocaleTimeString([], {hour: '2-digit', minute:'2-digit', second:'2-digit'})}] ${text}</div>`;
            box.scrollTop = box.scrollHeight;
        }

        
        function showInsufficientMaterials() {
            const el = document.getElementById('insufficient-materials');
            if(!el) return;
            el.classList.remove('hidden');
            clearTimeout(window.insufficientMatTimer);
            window.insufficientMatTimer = setTimeout(()=>el.classList.add('hidden'),1500);
        }

function switchTab(tab) {

            document.querySelectorAll('[id^="tab-"]').forEach(el => {
                el.classList.remove('text-amber-500', 'bg-stone-900');
                el.classList.add('text-stone-400', 'hover:text-stone-200');

            document.getElementById(`tab-${tab}`).classList.add('text-amber-500', 'bg-stone-900');
            document.getElementById(`tab-${tab}`).classList.remove('text-stone-400', 'hover:text-stone-200');

            document.querySelectorAll('[id^="panel-"]').forEach(el => el.classList.add('hidden'));
            document.getElementById(`panel-${tab}`).classList.remove('hidden');
        }

        function saveGame() {
            try {
                const saveData = {
                    game,
                    mapObjects: mapObjects.map(obj => ({
                        id: obj.id,
                        type: obj.type,
                        x: obj.x, y: obj.y, z: obj.z,
                        size: obj.size, hp: obj.hp, maxHp: obj.maxHp,
                        height: obj.height || 0, quality: obj.quality || 0,
                        animalType: obj.animalType || null
                    }))
                };
                localStorage.setItem('apocalypse_survival_save', JSON.stringify(saveData));
                logMsg("進度已成功儲存至本地瀏覽器！", "success");
                showModal("💾 存檔成功", "你的生存進度已儲存。");
            } catch (e) {
                logMsg("存檔失敗: " + e.message, "danger");
            }
        }

        function loadGame() {
            try {
                const dataStr = localStorage.getItem('apocalypse_survival_save');
                if (!dataStr) return;
                const saveData = JSON.parse(dataStr);
                
                game = saveData.game;
                if (!game.mode) game.mode = '365';
                
                mapObjects = [];
                saveData.mapObjects.forEach(obj => {
                    let newObj = createMapObject(obj.type, obj.x, obj.y, obj.z, obj.height);
                    newObj.hp = obj.hp; newObj.maxHp = obj.maxHp; newObj.quality = obj.quality || 0;
                    if(obj.animalType) newObj.animalType = obj.animalType;
                    mapObjects.push(newObj);
                });

                logMsg("已載入儲存進度！", "success");
                updateUI();
            } catch (e) {
                logMsg("讀檔失敗，存檔已損壞: " + e.message, "danger");
            }
        }
