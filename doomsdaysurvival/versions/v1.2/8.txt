                        
                        if (distToPlayer < 30) {
                            if (!obj.lastAttackTime || performance.now() - obj.lastAttackTime > 1200) {
                                game.hp = Math.max(0, game.hp - 8);
                                playSound('freeze');
                                logMsg("💥 洞穴怪物撕咬了你！生命值 -8", "danger");
                                obj.lastAttackTime = performance.now();
                                updateUI();
                            }
                        }
                    } else {
                        obj.animalWalkTimer += dt;
                        if (obj.animalWalkTimer > 3.0) { obj.animalWalkDir = Math.random() * Math.PI * 2; obj.animalWalkTimer = 0; }
                        if (Math.random() > 0.6) {
                            // 怪物生成於 10,000 px 地圖邊界內
                            obj.x = Math.max(WORLD_SIZE - 800, Math.min(WORLD_SIZE, obj.x + Math.cos(obj.animalWalkDir) * 0.8));
                            obj.y = Math.max(WORLD_SIZE - 800, Math.min(WORLD_SIZE, obj.y + Math.sin(obj.animalWalkDir) * 0.8));
                        }
                    }
                }
            });

            for (let i = bullets.length - 1; i >= 0; i--) {
                let b = bullets[i];
                b.x += b.vx;
                b.y += b.vy;
                b.life--;

                let hitObj = null;
                for (let j = 0; j < mapObjects.length; j++) {
                    let obj = mapObjects[j];
                    if (obj.type === 'animal' || obj.type === 'monster') {
                        if (Math.hypot(b.x - obj.x, b.y - obj.y) < obj.size + 5) {
                            hitObj = obj;
                            break;
                        }
                    }
                }

                if (hitObj) {
                    let baseDmg = b.type === 'm4a1' ? 25 : 55;
                    let headshotDmg = b.type === 'm4a1' ? 35 : 150;
                    let isHeadshot = Math.random() < 0.25; 
                    let finalDmg = isHeadshot ? headshotDmg : baseDmg;

                    hitObj.hp -= finalDmg;
                    createParticles(hitObj.x, hitObj.y, '#ef4444', 6);
                    
                    if (isHeadshot) {
                        showFloatText(hitObj.x, hitObj.y - 15, `HEADSHOT! -${finalDmg}`, '#f43f5e');
                        playSound('chime');
                    } else {
                        showFloatText(hitObj.x, hitObj.y - 15, `-${finalDmg}`, '#f87171');
                    }

                    if (hitObj.hp <= 0) {
                        let cap = getBackpackCapacity();
                        let current = getTotalItemsCount();

                        if (hitObj.type === 'animal') {
                            let gotMeat = Math.floor(2 + Math.random() * 3);
                            let gotLeather = Math.floor(1 + Math.random() * 2);
                            
                            if (current + gotMeat > cap) gotMeat = Math.max(0, cap - current);
                            game.resources.rawMeat += gotMeat;
                            current = getTotalItemsCount();
                            
                            if (current + gotLeather > cap) gotLeather = Math.max(0, cap - current);
                            game.resources.leather += gotLeather;

                            logMsg(`⚔️ 射殺獵物，獲得生肉 x${gotMeat}，皮革 x${gotLeather}`, "success");
                            mapObjects = mapObjects.filter(o => o.id !== hitObj.id);
                            setTimeout(() => { mapObjects.push(createMapObject('animal', Math.random()*WORLD_SIZE, Math.random()*WORLD_SIZE)); }, 5000);
                        } 
                        else if (hitObj.type === 'monster') {
                            let gotGunpowder = Math.floor(2 + Math.random() * 3); 
                            if (current + gotGunpowder > cap) gotGunpowder = Math.max(0, cap - current);
                            game.resources.gunpowder += gotGunpowder;

                            logMsg(`⚔️ 擊殺洞穴惡魔，獲得火藥 x${gotGunpowder}！`, "success");
                            mapObjects = mapObjects.filter(o => o.id !== hitObj.id);
                            setTimeout(() => { mapObjects.push(createMapObject('monster', (WORLD_SIZE - 500) + Math.random()*450, (WORLD_SIZE - 500) + Math.random()*450)); }, 8000);
                        }
                    }
                    bullets.splice(i, 1);
                    updateUI();
                } else if (b.life <= 0) {
                    bullets.splice(i, 1);
                }
            }

            for (let i = particles.length - 1; i >= 0; i--) {
                let p = particles[i];
                p.x += p.vx; p.y += p.vy; p.life--;
                if (p.life <= 0) particles.splice(i, 1);
            }

            for (let i = floatTexts.length - 1; i >= 0; i--) {
                let ft = floatTexts[i];
                ft.y -= 1; ft.life--;
                if (ft.life <= 0) floatTexts.splice(i, 1);
            }
        }

        // --- 寒冷判定、時間更新與致命判定 ---
        function updateGameTime(dt) {
            game.timeOfDay += dt; 
            
            let monsterCount = mapObjects.filter(o => o.type === 'monster').length;
            if (monsterCount < 6 && Math.random() < 0.03) {
                mapObjects.push(createMapObject('monster', (WORLD_SIZE - 500) + Math.random()*450, (WORLD_SIZE - 500) + Math.random()*450));
            }

            if (game.timeOfDay >= 600) { 
                game.timeOfDay = 0;
                game.day++;
                logMsg(`🌙 新的一天到來！第 ${game.day} 天。`, "info");
                
                game.hunger = Math.max(0, game.hunger - 15);
                if (game.hunger <= 0) {
                    game.hp = Math.max(0, game.hp - 20);
                    logMsg("🔥 你飢寒交迫！因嚴重飢餓而虛脫扣血！", "danger");
                }

                if (game.shelter.upgrading) {
                    game.shelter.daysRemaining--;
                    if (game.shelter.daysRemaining <= 0) {
                        game.shelter.upgrading = false;
                        game.shelter.level = game.shelter.targetLevel;
                        game.shelter.built = true;
                        logMsg(`🎉 你的庇護所已被成功升級/建造完成至 LV.${game.shelter.level}！`, "success");
                        showModal("🎉 升級完成", `你的庇護所已建造至 ${game.shelter.level} 等級！現在更溫暖抗寒了。`);
                    }
                }

                if (game.mode !== 'endless' && game.day > parseInt(game.mode)) {
                    isGameRunning = false;
                    showModal("🏆 生存挑戰成功！", `你成功在末日極低溫世界中活過了 ${game.mode} 天！恭喜你！`);
                    startGame(false);
                }

                updateUI();
            }
        }

        function updateStatusBars(dt) {
            let nearFurnace = false;
            let nearWorkbench = false;
            mapObjects.forEach(obj => {
                if (obj.type === 'furnace' && Math.hypot(game.player.x - obj.x, game.player.y - obj.y) < 100) nearFurnace = true;
                if (obj.type === 'workbench' && Math.hypot(game.player.x - obj.x, game.player.y - obj.y) < 70) nearWorkbench = true;
            });

            // 工作台接近自動彈出與遠離自動關閉 UI 機制
            if (nearWorkbench) {
                if (!game.wasNearWorkbench && !game.workbenchManuallyClosed) {
                    openWorkbenchUI();
                }
            } else {
                if (game.wasNearWorkbench) {
                    closeWorkbenchUI();
                    // 遠離時也確保購買彈窗被關閉
                    document.getElementById('craft-confirm-modal').classList.add('hidden');
                }
                game.workbenchManuallyClosed = false; 
            }
            game.wasNearWorkbench = nearWorkbench;

            let isNight = game.timeOfDay >= 300;
            let hasWarmth = nearFurnace || (game.shelter.built && game.shelter.level >= 0);
            let clothesProtection = game.equippedClothes ? game.equippedClothes.warmth : 0.0;
            const tempLabel = document.getElementById('temp-status');
            const warningDiv = document.getElementById('freeze-warning');

            if (isNight && !hasWarmth) {
                tempLabel.innerText = "極寒 (-15°C)"; tempLabel.className = "font-bold text-sky-400 animate-pulse";
                warningDiv.classList.remove('hidden');
                let rawDamage = exactFreezeDamage * (1.0 - clothesProtection);
                let finalDmg = Math.max(1, Math.round(rawDamage));
                game.hp = Math.max(0, game.hp - finalDmg);
                playSound('freeze');
                exactFreezeDamage *= 1.1;
                document.getElementById('freeze-dmg-show').innerText = Math.round(exactFreezeDamage);
            } else {
                tempLabel.innerText = isNight ? "涼爽 (10°C)" : "舒適 (24°C)";
                tempLabel.className = "font-bold text-emerald-400";
                warningDiv.classList.add('hidden');
                exactFreezeDamage = 2.0;
            }

            if (game.hp <= 0) {
                isGameRunning = false;
                showModal("💀 你死去了", `你未能捱過末日考驗。存活了 ${game.day} 天。`);
                startGame(false);
            }
            updateUI();
        }

        function performAction() {
            if (!isGameRunning) return false;
            
            if (game.equippedWeapon !== 'none') {
                fireBullet();
                return true;
            }

            let nearestObj = null; let minDist = 75;
            mapObjects.forEach(obj => {
                let d = Math.hypot(game.player.x - obj.x, game.player.y - obj.y);
                if (d < minDist) { nearestObj = obj; minDist = d; }
            });

            if (nearestObj) {
                let currentTotal = getTotalItemsCount();
                let backpackCap = getBackpackCapacity();
                if (currentTotal >= backpackCap && (nearestObj.type === 'tree' || nearestObj.type === 'steel_pile' || nearestObj.type === 'leather_pile' || nearestObj.type === 'animal')) {
                    showModal("🎒 背包已滿！", `你目前的物品數為 ${currentTotal}/${backpackCap}，無法繼續收集資源，請升級背包或消耗物資。`);
                    return false;
                }

                if (nearestObj.type === 'tree') {
                    playSound('chop');
                    let power = game.equippedAxe ? game.equippedAxe.efficiency : 1;
                    nearestObj.hp -= power;
                    let got = Math.floor(2 + Math.random() * 3) * power;
                    if (currentTotal + got > backpackCap) got = Math.max(0, backpackCap - currentTotal);
                    game.resources.wood += got; logMsg(`🪓 砍樹，獲得木頭 x${got}`, "success");
                    if (nearestObj.hp <= 0) {
                        mapObjects = mapObjects.filter(o => o.id !== nearestObj.id);
                        setTimeout(() => { mapObjects.push(createMapObject('tree', Math.random()*WORLD_SIZE, Math.random()*WORLD_SIZE)); }, 3000);
                    }
                    updateUI(); return true;
                }
                else if (nearestObj.type === 'steel_pile') {
                    playSound('hit');
                    nearestObj.hp--;
                    let got = Math.floor(1 + Math.random() * 2);
                    if (currentTotal + got > backpackCap) got = Math.max(0, backpackCap - currentTotal);
                    game.resources.steel += got; logMsg(`⛏️ 採集廢鋼堆，獲得鋼材 x${got}`, "success");
                    if (nearestObj.hp <= 0) { mapObjects = mapObjects.filter(o => o.id !== nearestObj.id); }
                    updateUI(); return true;
                }
                else if (nearestObj.type === 'leather_pile') {
                    playSound('hit');
                    nearestObj.hp--;
                    let got = Math.floor(1 + Math.random() * 3);
                    if (currentTotal + got > backpackCap) got = Math.max(0, backpackCap - currentTotal);
                    game.resources.leather += got; logMsg(`🧶 搜刮皮卷，獲得皮革 x${got}`, "success");
