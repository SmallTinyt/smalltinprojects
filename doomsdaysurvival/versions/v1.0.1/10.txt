        // --- 小地圖繪製與互動 ---
        const mapCanvas = document.getElementById('minimap-canvas');
        mapCanvas.addEventListener('click', (e) => {
            const rect = mapCanvas.getBoundingClientRect();
            const mx = e.clientX - rect.left;
            const my = e.clientY - rect.top;
            const scale = WORLD_SIZE / mapCanvas.width;
            game.customMarker = { x: mx * scale, y: my * scale };
            drawMinimap();
        });

        function drawMinimap() {
            const mctx = mapCanvas.getContext('2d');
            mctx.clearRect(0, 0, mapCanvas.width, mapCanvas.height);
            const scale = mapCanvas.width / WORLD_SIZE;

            // 格線
            mctx.strokeStyle = 'rgba(255,255,255,0.05)';
            for(let i=0; i<WORLD_SIZE; i+=280) {
                mctx.beginPath(); mctx.moveTo(i*scale, 0); mctx.lineTo(i*scale, mapCanvas.height); mctx.stroke();
                mctx.beginPath(); mctx.moveTo(0, i*scale); mctx.lineTo(mapCanvas.width, i*scale); mctx.stroke();
            }

            // 實體點
            mapObjects.forEach(obj => {
                let size = 2;
                if (obj.type === 'building') { mctx.fillStyle = '#475569'; size = 12; }
                else if (obj.type === 'tree') mctx.fillStyle = '#15803d';
                else if (obj.type === 'workbench') { mctx.fillStyle = '#3b82f6'; size = 6; }
                else if (obj.type === 'shelter') { mctx.fillStyle = '#d97706'; size = 8; }
                else return; // 其他不畫
                
                mctx.fillRect(obj.x * scale - size/2, obj.y * scale - size/2, size, size);
            });

            // 自訂標記
            if (game.customMarker) {
                mctx.fillStyle = '#ef4444';
                mctx.beginPath();
                mctx.arc(game.customMarker.x * scale, game.customMarker.y * scale, 4, 0, Math.PI*2);
                mctx.fill();
            }

            // 玩家
            mctx.fillStyle = '#22c55e';
            mctx.beginPath();
            mctx.arc(game.player.x * scale, game.player.y * scale, 4, 0, Math.PI*2);
            mctx.fill();
        }
