            } else if (game.shelter.upgrading) {
                shelterTitle.innerText = `庇護所 施工中... ➔ LV.${game.shelter.targetLevel}`;
                shelterBtn.setAttribute('disabled', 'true'); shelterBtn.innerText = "施工中...";
                document.getElementById('shelter-progress-container').classList.remove('hidden');
                let t = game.shelter.targetLevel; let total = t===1?1:(t===2?3:(t===3?5:10));
                document.getElementById('shelter-progress-bar').style.width = `${((total - game.shelter.daysRemaining) / total) * 100}%`;
                document.getElementById('shelter-timer').innerText = `剩餘: ${game.shelter.daysRemaining} 天`;
            } else {
                document.getElementById('shelter-icon').innerText = game.shelter.level >= 1 ? "🏡" : "⛺";
                shelterTitle.innerText = `庇護所 (LV.${game.shelter.level})`; document.getElementById('shelter-progress-container').classList.add('hidden');
                shelterBtn.removeAttribute('disabled');
                let next = game.shelter.level + 1; let cw = 30 * Math.pow(2, next), cs = 20 * Math.pow(2, next);
                shelterDesc.innerText = `升至 LV.${next} 需：木頭 x${cw}, 鋼鐵 x${cs}`; shelterBtn.innerText = `升級至 ${next} 級`;
            }
        }

        requestAnimationFrame(gameLoop);
