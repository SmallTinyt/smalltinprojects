                        type: obj.type, x: obj.x, y: obj.y, z: obj.z,
                        size: obj.size, hp: obj.hp, maxHp: obj.maxHp,
                        height: obj.height || 0, quality: obj.quality || 0,
                        animalType: obj.animalType || null
                    }))
                };
                localStorage.setItem('apocalypse_survival_save', JSON.stringify(saveData));
                logMsg("進度已成功儲存至本地瀏覽器！", "success");
                showModal("💾 存檔成功", "你的生存進度已儲存。");
            } catch (e) {
                logMsg("存檔失敗: " + e.message, "danger");
            }
        }

        function loadGame() {
            try {
                const dataStr = localStorage.getItem('apocalypse_survival_save');
                if (!dataStr) return;
                const saveData = JSON.parse(dataStr);
                
                game = saveData.game;
                // 向下相容預設值
                if (!game.mode) game.mode = '365';
                
                mapObjects = [];
                saveData.mapObjects.forEach(obj => {
                    let newObj = createMapObject(obj.type, obj.x, obj.y, obj.z, obj.height);
                    newObj.hp = obj.hp; newObj.maxHp = obj.maxHp; newObj.quality = obj.quality || 0;
                    if(obj.animalType) newObj.animalType = obj.animalType;
                    mapObjects.push(newObj);
                });

                logMsg("成功讀取上次的存檔進度！", "success");
                updateUI();
            } catch (e) {
                logMsg("讀取存檔錯誤: " + e.message, "danger");
            }
        }

        function eatMeat(type) {
            if (type === 'raw') {
                if (game.resources.rawMeat <= 0) return;
                game.resources.rawMeat--;
                game.hunger = Math.min(game.maxHunger, game.hunger + 10);
                game.hp = Math.max(1, game.hp - 5);
                logMsg("吃了生肉。(生命 -5, 飢餓 +10)", "warning");
                playSound('hit');
            } else if (type === 'cooked') {
                if (game.resources.cookedMeat <= 0) return;
                game.resources.cookedMeat--;
                game.hunger = Math.min(game.maxHunger, game.hunger + 40);
                game.hp = Math.min(game.maxHp, game.hp + 10);
                logMsg("享用了美味的烤熟肉！(生命 +10, 飢餓 +40)", "success");
                playSound('chime');
            }
            updateUI();
        }

        function createMapObject(type, x, y, z = 0, height = 0) {
            let size = 20, hp = 10;
            if (type === 'tree') { size = 24; hp = 5; }
            else if (type === 'steel_pile') { size = 18; hp = 8; }
            else if (type === 'leather_pile') { size = 16; hp = 4; }
            else if (type === 'building') { size = 60; hp = 999999; }
            else if (type === 'animal') { size = 15; hp = 3; }
            else if (type === 'workbench') { size = 25; hp = 9999; }
            else if (type === 'furnace') { size = 25; hp = 9999; }
            else if (type === 'bed') { size = 30; hp = 9999; }
            else if (type === 'shelter') { size = 45; hp = 9999; }

            return {
                id: Math.random().toString(36).substr(2, 9),
                type, x, y, z, size, hp, maxHp: hp, height, quality: 0,
                animalType: type === 'animal' ? (Math.random() > 0.5 ? 'cow' : 'pig') : null,
                animalWalkDir: Math.random() * Math.PI * 2, animalWalkTimer: 0
            };
        }

        function initMap() {
            mapObjects = [];
            for (let x = 100; x < WORLD_SIZE; x += 280) {
                for (let y = 100; y < WORLD_SIZE; y += 280) {
                    if (Math.abs(x - 400) < 150 && Math.abs(y - 400) < 150) continue;
                    mapObjects.push(createMapObject('building', x, y, 0, 80 + Math.random() * 160));
                }
            }
            mapObjects.push(createMapObject('workbench', 450, 380, 0));
