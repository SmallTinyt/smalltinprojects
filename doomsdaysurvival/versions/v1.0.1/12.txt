                    if (Math.hypot(game.player.x - obj.x, game.player.y - obj.y) < obj.size * 0.8) {
                        game.player.x = oldX; game.player.y = oldY;
                    }
                }
            });

            game.player.x = Math.max(20, Math.min(WORLD_SIZE - 20, game.player.x));
            game.player.y = Math.max(20, Math.min(WORLD_SIZE - 20, game.player.y));

            let distMoved = Math.hypot(game.player.x - oldX, game.player.y - oldY);
            if (distMoved > 0.05) {
                stepCountReset += distMoved;
                if (stepCountReset >= 16) {
                    game.player.stepsCount++; stepCountReset = 0;
                    if (game.player.stepsCount >= 10) {
                        game.stamina = Math.max(0, game.stamina - 1);
                        game.player.stepsCount = 0; updateUI();
                    }
                }
            }

            mapObjects.forEach(obj => {
                if (obj.type === 'animal') {
                    obj.animalWalkTimer += dt;
                    if (obj.animalWalkTimer > 2.5) { obj.animalWalkDir = Math.random() * Math.PI * 2; obj.animalWalkTimer = 0; }
                    obj.x += Math.cos(obj.animalWalkDir) * 0.4; obj.y += Math.sin(obj.animalWalkDir) * 0.4;
                    obj.x = Math.max(10, Math.min(WORLD_SIZE - 10, obj.x)); obj.y = Math.max(10, Math.min(WORLD_SIZE - 10, obj.y));
                }
            });
            
            game.player.vx = mx; game.player.vy = my;
        }

        function updateGameTime(dt) {
            game.timeOfDay += dt;
            if (game.timeOfDay >= 600) {
                game.timeOfDay = 0;
                game.day++;
                logMsg(`🎉 新的一天開始了！當前為生存第 ${game.day} 天。`, "success");
                playSound('chime');

                if (game.shelter.upgrading) {
                    game.shelter.daysRemaining--;
                    if (game.shelter.daysRemaining <= 0) {
                        game.shelter.built = true; game.shelter.level = game.shelter.targetLevel; game.shelter.upgrading = false;
                        logMsg(`🏆 庇護所成功升級到等級 ${game.shelter.level}！`, "success");
                        showModal("⛺ 庇護所升級完成", `庇護所已升級為 LV.${game.shelter.level}！這將提供更優異的保護。`);
                    } else {
                        logMsg(`🔨 庇護所施工剩餘 ${game.shelter.daysRemaining} 天。`, "success");
                    }
                }

                // 檢查獲勝條件
                if (game.mode !== 'endless' && game.day > parseInt(game.mode)) {
                    isGameRunning = false;
                    showModal("🏆 恭喜通關！", `你成功在末日都市裡存活了 ${game.mode} 天！你是真正的生存大師！`);
                }
            }
        }
