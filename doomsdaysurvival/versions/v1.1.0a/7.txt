                else if (game.player.facing === 'SW') angle = 3 * Math.PI / 4;
                else angle = Math.PI / 4;
            }

            bullets.push({
                x: game.player.x,
                y: game.player.y,
                vx: Math.cos(angle) * 16,
                vy: Math.sin(angle) * 16,
                type: game.equippedWeapon,
                life: 50 
            });

            playGunSound(game.equippedWeapon);
            updateUI();
        }

        // 隨機生成飄字特效
        let floatTexts = [];
        function showFloatText(x, y, text, color) {
            floatTexts.push({ x: x, y: y, text: text, color: color, life: 60, maxLife: 60 });
        }

        function createParticles(x, y, color, count) {
            for(let i=0; i<count; i++) {
                particles.push({
                    x: x, y: y,
                    vx: (Math.random() - 0.5) * 6,
                    vy: (Math.random() - 0.5) * 6,
                    life: Math.random() * 20 + 10,
                    color: color,
                    size: Math.random() * 4 + 2
                });
            }
        }
        let particles = [];

        // --- 核心更新循環與物理碰撞 ---
        function gameLoop(now) {
            let dt = (now - lastTime) / 1000;
            if (dt > 0.1) dt = 0.1;
            lastTime = now;

            if (isGameRunning && !isMapOpen) {
                updatePhysics(dt);
                updateGameTime(dt);
                updateStatusBars(dt);
                render();
            }

            requestAnimationFrame(gameLoop);
        }

        function updatePhysics(dt) {
            let mx = 0, my = 0;
            if (keys['w'] || keys['arrowup']) { my = -1; game.player.facing = 'NW'; }
            if (keys['s'] || keys['arrowdown']) { my = 1; game.player.facing = 'SE'; }
            if (keys['a'] || keys['arrowleft']) { mx = -1; game.player.facing = 'SW'; }
            if (keys['d'] || keys['arrowright']) { mx = 1; game.player.facing = 'NE'; }

            if (isJoystickActive) {
                mx = joystickVector.x; my = joystickVector.y;
                if (Math.abs(mx) > Math.abs(my)) game.player.facing = mx > 0 ? 'NE' : 'SW';
                else game.player.facing = my > 0 ? 'SE' : 'NW';
            }

            if (mx !== 0 && my !== 0) { const len = Math.hypot(mx, my); mx /= len; my /= len; }

            let speed = game.player.speed;
            let oldX = game.player.x;
            let oldY = game.player.y;

            game.player.vx = mx * speed;
            game.player.vy = my * speed;

            game.player.x += game.player.vx;
            game.player.y += game.player.vy;

            mapObjects.forEach(obj => {
                if (obj.type === 'building' || obj.type === 'shelter' || obj.type === 'cave') {
                    if (Math.hypot(game.player.x - obj.x, game.player.y - obj.y) < obj.size * 0.8) {
                        game.player.x = oldX; game.player.y = oldY;
                    }
                }
            });

            game.player.x = Math.max(20, Math.min(WORLD_SIZE - 20, game.player.x));
            game.player.y = Math.max(20, Math.min(WORLD_SIZE - 20, game.player.y));

            let distMoved = Math.hypot(game.player.x - oldX, game.player.y - oldY);
            if (distMoved > 0.05) {
                stepCountReset += distMoved;
                if (stepCountReset >= 16) {
                    game.player.stepsCount++; stepCountReset = 0;
                    if (game.player.stepsCount >= 10) {
                        game.stamina = Math.max(0, game.stamina - 1);
                        game.player.stepsCount = 0; updateUI();
                    }
                }
            }

            mapObjects.forEach(obj => {
                if (obj.type === 'animal') {
                    obj.animalWalkTimer += dt;
                    if (obj.animalWalkTimer > 2.5) { obj.animalWalkDir = Math.random() * Math.PI * 2; obj.animalWalkTimer = 0; }
                    
                    if (Math.random() > 0.4) {
                        let ax = Math.cos(obj.animalWalkDir) * 1.0;
                        let ay = Math.sin(obj.animalWalkDir) * 1.0;
                        obj.x = Math.max(20, Math.min(WORLD_SIZE - 20, obj.x + ax));
                        obj.y = Math.max(20, Math.min(WORLD_SIZE - 20, obj.y + ay));
                    }
                } 
                else if (obj.type === 'monster') {
                    let distToPlayer = Math.hypot(game.player.x - obj.x, game.player.y - obj.y);
                    if (distToPlayer < 400) { 
                        let angle = Math.atan2(game.player.y - obj.y, game.player.x - obj.x);
                        obj.x += Math.cos(angle) * 1.6; 
                        obj.y += Math.sin(angle) * 1.6;
                        
                        if (distToPlayer < 30) {
                            if (!obj.lastAttackTime || performance.now() - obj.lastAttackTime > 1200) {
                                game.hp = Math.max(0, game.hp - 8);
                                playSound('freeze');
                                logMsg("💥 洞穴怪物撕咬了你！生命值 -8", "danger");
                                obj.lastAttackTime = performance.now();
                                updateUI();
                            }
                        }
                    } else {
                        obj.animalWalkTimer += dt;
                        if (obj.animalWalkTimer > 3.0) { obj.animalWalkDir = Math.random() * Math.PI * 2; obj.animalWalkTimer = 0; }
                        if (Math.random() > 0.6) {
                            // 怪物生成於 10,000 px 地圖邊界內
                            obj.x = Math.max(WORLD_SIZE - 800, Math.min(WORLD_SIZE, obj.x + Math.cos(obj.animalWalkDir) * 0.8));
                            obj.y = Math.max(WORLD_SIZE - 800, Math.min(WORLD_SIZE, obj.y + Math.sin(obj.animalWalkDir) * 0.8));
                        }
                    }
                }
            });

            for (let i = bullets.length - 1; i >= 0; i--) {
                let b = bullets[i];
                b.x += b.vx;
                b.y += b.vy;
                b.life--;

                let hitObj = null;
                for (let j = 0; j < mapObjects.length; j++) {
                    let obj = mapObjects[j];
                    if (obj.type === 'animal' || obj.type === 'monster') {
                        if (Math.hypot(b.x - obj.x, b.y - obj.y) < obj.size + 5) {
                            hitObj = obj;
                            break;
                        }
                    }
                }

                if (hitObj) {
                    let baseDmg = b.type === 'm4a1' ? 25 : 55;
                    let headshotDmg = b.type === 'm4a1' ? 35 : 150;
                    let isHeadshot = Math.random() < 0.25; 
                    let finalDmg = isHeadshot ? headshotDmg : baseDmg;

                    hitObj.hp -= finalDmg;
                    createParticles(hitObj.x, hitObj.y, '#ef4444', 6);
                    
                    if (isHeadshot) {
                        showFloatText(hitObj.x, hitObj.y - 15, `HEADSHOT! -${finalDmg}`, '#f43f5e');
                        playSound('chime');
                    } else {
                        showFloatText(hitObj.x, hitObj.y - 15, `-${finalDmg}`, '#f87171');
                    }

                    if (hitObj.hp <= 0) {
                        let cap = getBackpackCapacity();
                        let current = getTotalItemsCount();

                        if (hitObj.type === 'animal') {
                            let gotMeat = Math.floor(2 + Math.random() * 3);
                            let gotLeather = Math.floor(1 + Math.random() * 2);
                            
                            if (current + gotMeat > cap) gotMeat = Math.max(0, cap - current);
                            game.resources.rawMeat += gotMeat;
                            current = getTotalItemsCount();
                            
                            if (current + gotLeather > cap) gotLeather = Math.max(0, cap - current);
                            game.resources.leather += gotLeather;

                            logMsg(`⚔️ 射殺獵物，獲得生肉 x${gotMeat}，皮革 x${gotLeather}`, "success");
                            mapObjects = mapObjects.filter(o => o.id !== hitObj.id);
                            setTimeout(() => { mapObjects.push(createMapObject('animal', Math.random()*WORLD_SIZE, Math.random()*WORLD_SIZE)); }, 5000);
                        } 
                        else if (hitObj.type === 'monster') {
                            let gotGunpowder = Math.floor(2 + Math.random() * 3); 
                            if (current + gotGunpowder > cap) gotGunpowder = Math.max(0, cap - current);
                            game.resources.gunpowder += gotGunpowder;

                            logMsg(`⚔️ 擊殺洞穴惡魔，獲得火藥 x${gotGunpowder}！`, "success");
                            mapObjects = mapObjects.filter(o => o.id !== hitObj.id);
                            setTimeout(() => { mapObjects.push(createMapObject('monster', (WORLD_SIZE - 500) + Math.random()*450, (WORLD_SIZE - 500) + Math.random()*450)); }, 8000);
                        }
                    }
                    bullets.splice(i, 1);
                    updateUI();
                } else if (b.life <= 0) {
                    bullets.splice(i, 1);
                }
            }

            for (let i = particles.length - 1; i >= 0; i--) {
                let p = particles[i];
                p.x += p.vx; p.y += p.vy; p.life--;
                if (p.life <= 0) particles.splice(i, 1);
            }

            for (let i = floatTexts.length - 1; i >= 0; i--) {
                let ft = floatTexts[i];
                ft.y -= 1; ft.life--;
                if (ft.life <= 0) floatTexts.splice(i, 1);
            }
        }

        // --- 寒冷判定、時間更新與致命判定 ---
        function updateGameTime(dt) {
            game.timeOfDay += dt; 
            
            let monsterCount = mapObjects.filter(o => o.type === 'monster').length;
            if (monsterCount < 6 && Math.random() < 0.03) {
                mapObjects.push(createMapObject('monster', (WORLD_SIZE - 500) + Math.random()*450, (WORLD_SIZE - 500) + Math.random()*450));
            }

            if (game.timeOfDay >= 600) { 
                game.timeOfDay = 0;
                game.day++;
                logMsg(`🌙 新的一天到來！第 ${game.day} 天。`, "info");
                
                game.hunger = Math.max(0, game.hunger - 15);
                if (game.hunger <= 0) {
                    game.hp = Math.max(0, game.hp - 20);
                    logMsg("🔥 你飢寒交迫！因嚴重飢餓而虛脫扣血！", "danger");
                }

                if (game.shelter.upgrading) {
                    game.shelter.daysRemaining--;
                    if (game.shelter.daysRemaining <= 0) {
                        game.shelter.upgrading = false;
                        game.shelter.level = game.shelter.targetLevel;
                        game.shelter.built = true;
                        logMsg(`🎉 你的庇護所已被成功升級/建造完成至 LV.${game.shelter.level}！`, "success");
