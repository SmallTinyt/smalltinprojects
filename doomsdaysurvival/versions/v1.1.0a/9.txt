            else if (recipeId === 'bed_2') { game.equippedBed = { name: "精工軟彈簧床", quality: 2, staminaRate: 20 }; }
            else if (recipeId === 'backpack_1') { game.backpackLvl = 1; }
            else if (recipeId === 'backpack_2') { game.backpackLvl = 2; }
            else if (recipeId === 'backpack_3') { game.backpackLvl = 3; }
            else if (recipeId === 'm4a1') { game.resources.m4a1Count = (game.resources.m4a1Count || 0) + 1; }
            else if (recipeId === 'm4a1_ammo') { game.resources.m4a1Ammo = (game.resources.m4a1Ammo || 0) + 30; }
            else if (recipeId === 'revolver') { game.resources.revolverCount = (game.resources.revolverCount || 0) + 1; }
            else if (recipeId === 'revolver_ammo') { game.resources.revolverAmmo = (game.resources.revolverAmmo || 0) + 10; }
            else if (recipeId === 'furnace') {
                mapObjects.push(createMapObject('furnace', game.player.x + 30, game.player.y + 30));
                logMsg("🔥 已在地上架設了野外保暖熔爐！", "success");
            }

            playSound('craft'); logMsg(`🎉 成功合成 ${recipe.name}!`, "success"); 
            updateWorkbenchOverlayUI(); // 即時重繪工作台確保狀態更新
            updateUI();
        }

        function eatMeat(type) {
            if (type === 'raw') {
                if (game.resources.rawMeat > 0) {
                    game.resources.rawMeat--;
                    game.hunger = Math.min(100, game.hunger + 10);
                    game.hp = Math.max(5, game.hp - 10);
                    logMsg("🥩 吞下了生肉！口感極差，恢復微量飢餓並扣減10生命值。", "danger");
                }
            } else {
                if (game.resources.cookedMeat > 0) {
                    game.resources.cookedMeat--;
                    game.hunger = Math.min(100, game.hunger + 40);
                    game.hp = Math.min(100, game.hp + 5);
                    logMsg("🍗 吃了美味熟肉，感覺精力充沛！飢餓值大幅恢復。", "success");
                }
            }
            updateUI();
        }

        function resizeCanvas() {
            const targetRatio = 16 / 9;
            let w = window.innerWidth;
            let h = window.innerHeight;
            
            if (w / h > targetRatio) {
                canvas.height = h;
                canvas.width = h * targetRatio;
            } else {
                canvas.width = w;
                canvas.height = w / targetRatio;
            }
        }
        window.addEventListener('resize', resizeCanvas);
