                    
                    if (pos.x < -100 || pos.x > canvas.width + 100 || pos.y < -100 || pos.y > canvas.height + 150) return;

                    if (obj.type === 'tree') drawTree(pos.x, pos.y, ambientLight);
                    else if (obj.type === 'steel_pile') drawSteelPile(pos.x, pos.y, ambientLight);
                    else if (obj.type === 'leather_pile') drawLeatherPile(pos.x, pos.y, ambientLight);
                    else if (obj.type === 'building') drawBuilding(pos.x, pos.y, obj.height, ambientLight);
                    else if (obj.type === 'cave') drawCave(pos.x, pos.y, ambientLight); 
                    else if (obj.type === 'monster') drawMonster(pos.x, pos.y, ambientLight); 
                    else if (obj.type === 'animal') drawAnimal(pos.x, pos.y, obj.animalType, obj.animalWalkDir, ambientLight);
                    else if (obj.type === 'workbench') drawIsoBox(ctx, pos.x, pos.y, 22, 14, 10, game.workbenchLvl === 1 ? '#ca8a04' : (game.workbenchLvl === 2 ? '#0284c7' : '#9333ea'), '#b45309', '#78350f');
                    else if (obj.type === 'furnace') drawFurnace(pos.x, pos.y, ambientLight);
                    else if (obj.type === 'bed') drawBed(pos.x, pos.y, obj, ambientLight);
                    else if (obj.type === 'shelter') drawShelter(pos.x, pos.y, obj, ambientLight);

                    if (obj.hp < obj.maxHp && obj.maxHp < 1000) {
                        ctx.fillStyle = 'rgba(0,0,0,0.6)'; ctx.fillRect(pos.x - 20, pos.y - 45, 40, 5);
                        ctx.fillStyle = '#ef4444'; ctx.fillRect(pos.x - 20, pos.y - 45, (obj.hp / obj.maxHp) * 40, 5);
                    }
                }
            });

            bullets.forEach(b => {
                let pos = worldToScreen(b.x, b.y);
                ctx.fillStyle = b.type === 'm4a1' ? '#f59e0b' : '#ef4444';
                ctx.beginPath();
                ctx.arc(pos.x, pos.y, 3, 0, Math.PI * 2);
                ctx.fill();
            });

            ctx.save();
            ctx.translate(-game.player.x * cos30, -game.player.y * sin30); // 獲取正確的世界轉換位置以明確標示世界邊界

            // ★ 明確標示物理世界 10,000 px 邊界：繪製厚實鮮紅色的警告線與內側半透明警告斜紋
            let screenBoundary = worldToScreen(0, 0);
            let screenEnd = worldToScreen(WORLD_SIZE, WORLD_SIZE);
            
            // 由於在 ISO 視角下用 World Coordinate 做投影最為準確
            ctx.restore();

            ctx.save();
            // 在 translated 座標系下繪製世界邊緣線
            const cx = canvas.width / 2;
            const cy = canvas.height / 2;
            ctx.translate(cx - game.player.x * cos30 + game.player.y * cos30, cy - game.player.x * sin30 - game.player.y * sin30);

            // 繪製正方形邊界的投影
            ctx.strokeStyle = '#f43f5e'; // 鮮紅色邊框
            ctx.lineWidth = 15;
            ctx.beginPath();
            let p1 = { x: (0 - 0)*cos30, y: (0 + 0)*sin30 };
            let p2 = { x: (WORLD_SIZE - 0)*cos30, y: (WORLD_SIZE + 0)*sin30 };
            let p3 = { x: (WORLD_SIZE - WORLD_SIZE)*cos30, y: (WORLD_SIZE + WORLD_SIZE)*sin30 };
            let p4 = { x: (0 - WORLD_SIZE)*cos30, y: (0 + WORLD_SIZE)*sin30 };
            ctx.moveTo(p1.x, p1.y);
            ctx.lineTo(p2.x, p2.y);
            ctx.lineTo(p3.x, p3.y);
            ctx.lineTo(p4.x, p4.y);
            ctx.closePath();
            ctx.stroke();

            // 繪製邊界內側警告斜紋 (Danger stripes)
            ctx.strokeStyle = 'rgba(244, 63, 94, 0.2)';
            ctx.lineWidth = 45;
            ctx.stroke();

            ctx.restore();

            if (ambientLight < 1.0) drawDarknessAndLights(ambientLight);

            drawEdgeIndicators();
        }

        // --- 繪製螢幕邊緣指示箭頭 ---
        function drawEdgeIndicators() {
            let targets = [];
            if (game.customMarker) targets.push({ x: game.customMarker.x, y: game.customMarker.y, color: '#ef4444', label: '標記' });
            
            let wb = mapObjects.find(o => o.type === 'workbench');
            if (wb) targets.push({ x: wb.x, y: wb.y, color: '#3b82f6', label: '工作台' });

            let cv = mapObjects.find(o => o.type === 'cave');
            if (cv) targets.push({ x: cv.x, y: cv.y, color: '#a855f7', label: '神秘洞穴副本' });
            
            if (game.shelter.level >= 0) {
                let sh = mapObjects.find(o => o.type === 'shelter');
                if (sh) targets.push({ x: sh.x, y: sh.y, color: '#d97706', label: '庇護所' });
            }

            targets.forEach(tgt => {
                let screenPos = worldToScreen(tgt.x, tgt.y);
                let edge = getEdgePos(screenPos.x, screenPos.y, 40);
                if (edge.offscreen) {
                    drawPointer(edge.x, edge.y, edge.angle, tgt.color, tgt.label);
                }
            });
        }

        function getEdgePos(px, py, padding) {
            let cx = canvas.width / 2;
            let cy = canvas.height / 2;
            let dx = px - cx;
            let dy = py - cy;
            let hw = cx - padding;
            let hh = cy - padding;

            if (Math.abs(dx) <= hw && Math.abs(dy) <= hh) return { offscreen: false };

            let scaleX = Math.abs(dx) > 0.001 ? hw / Math.abs(dx) : 9999;
            let scaleY = Math.abs(dy) > 0.001 ? hh / Math.abs(dy) : 9999;
            let scale = Math.min(scaleX, scaleY);

            return { x: cx + dx * scale, y: cy + dy * scale, offscreen: true, angle: Math.atan2(dy, dx) };
        }

        function drawPointer(x, y, angle, color, label) {
            ctx.save();
            ctx.translate(x, y);
            ctx.rotate(angle);
            ctx.fillStyle = color;
            ctx.beginPath();
            ctx.moveTo(12, 0); ctx.lineTo(-8, 8); ctx.lineTo(-8, -8); ctx.closePath();
            ctx.fill();
            ctx.shadowBlur = 8; ctx.shadowColor = color;
            ctx.strokeStyle = '#ffffff'; ctx.lineWidth = 1;
            ctx.stroke();
            ctx.restore();

            ctx.fillStyle = '#ffffff';
            ctx.font = 'bold 9px sans-serif';
            ctx.textAlign = 'center';
            ctx.fillText(label, x, y - 12);
        }

        // ★ 修改衛星雷達地圖：完全正方形，且觀測/坐標範圍鎖定在 3,000 px 規格內
        function drawMinimap() {
            mctx.fillStyle = '#0f172a';
            mctx.fillRect(0, 0, mapCanvas.width, mapCanvas.height);

            const MAP_RANGE = 3000; // 雷達地圖限額 3000 px
            let scale = mapCanvas.width / MAP_RANGE;

            // 繪製網格
            mctx.strokeStyle = '#1e293b'; mctx.lineWidth = 1;
            for (let i = 0; i <= MAP_RANGE; i += 500) {
                mctx.beginPath(); mctx.moveTo(i*scale, 0); mctx.lineTo(i*scale, mapCanvas.height); mctx.stroke();
                mctx.beginPath(); mctx.moveTo(0, i*scale); mctx.lineTo(mapCanvas.width, i*scale); mctx.stroke();
            }

            // 繪製物資實體
            mapObjects.forEach(obj => {
                // 如果超出 3000 px 的範圍則在地圖上裁剪不予顯示
                if (obj.x > MAP_RANGE || obj.y > MAP_RANGE) return;

                let size = 2;
                if (obj.type === 'building') { mctx.fillStyle = '#475569'; size = 8; }
                else if (obj.type === 'tree') mctx.fillStyle = '#15803d';
                else if (obj.type === 'cave') { mctx.fillStyle = '#a855f7'; size = 10; } 
                else if (obj.type === 'workbench') { mctx.fillStyle = '#3b82f6'; size = 5; }
                else if (obj.type === 'shelter') { mctx.fillStyle = '#d97706'; size = 7; }
                else return; 
                
                mctx.fillRect(obj.x * scale - size/2, obj.y * scale - size/2, size, size);
            });

            // 繪製自訂標記 (限額在 3000px 內才顯示)
            if (game.customMarker && game.customMarker.x <= MAP_RANGE && game.customMarker.y <= MAP_RANGE) {
                mctx.fillStyle = '#ef4444';
                mctx.beginPath();
                mctx.arc(game.customMarker.x * scale, game.customMarker.y * scale, 4, 0, Math.PI*2);
                mctx.fill();
            }

            // 繪製玩家位置 (限額在 3000px 內才顯示)
            if (game.player.x <= MAP_RANGE && game.player.y <= MAP_RANGE) {
                mctx.fillStyle = '#22c55e';
                mctx.beginPath();
                mctx.arc(game.player.x * scale, game.player.y * scale, 5, 0, Math.PI*2);
                mctx.fill();
            }
        }

        mapCanvas.addEventListener('mousedown', e => {
            let rect = mapCanvas.getBoundingClientRect();
            let clickX = e.clientX - rect.left;
            let clickY = e.clientY - rect.top;
            const MAP_RANGE = 3000;
            let scale = MAP_RANGE / mapCanvas.width;
            
            game.customMarker = { x: clickX * scale, y: clickY * scale };
            logMsg(`📌 設定了自訂地圖目標在: (${Math.round(game.customMarker.x)}, ${Math.round(game.customMarker.y)})`, "info");
            drawMinimap();
        });
