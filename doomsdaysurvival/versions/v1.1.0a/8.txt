                        showModal("🎉 升級完成", `你的庇護所已建造至 ${game.shelter.level} 等級！現在更溫暖抗寒了。`);
                    }
                }

                if (game.mode !== 'endless' && game.day > parseInt(game.mode)) {
                    isGameRunning = false;
                    showModal("🏆 生存挑戰成功！", `你成功在末日極低溫世界中活過了 ${game.mode} 天！恭喜你！`);
                    startGame(false);
                }

                updateUI();
            }
        }

        function updateStatusBars(dt) {
            let nearFurnace = false;
            let nearWorkbench = false;
            mapObjects.forEach(obj => {
                if (obj.type === 'furnace' && Math.hypot(game.player.x - obj.x, game.player.y - obj.y) < 100) nearFurnace = true;
                if (obj.type === 'workbench' && Math.hypot(game.player.x - obj.x, game.player.y - obj.y) < 70) nearWorkbench = true;
            });

            // 工作台接近自動彈出與遠離自動關閉 UI 機制
            if (nearWorkbench) {
                if (!game.wasNearWorkbench && !game.workbenchManuallyClosed) {
                    openWorkbenchUI();
                }
            } else {
                if (game.wasNearWorkbench) {
                    closeWorkbenchUI();
                    // 遠離時也確保購買彈窗被關閉
                    document.getElementById('craft-confirm-modal').classList.add('hidden');
                }
                game.workbenchManuallyClosed = false; 
            }
            game.wasNearWorkbench = nearWorkbench;

            let isNight = game.timeOfDay >= 300;
            let hasWarmth = nearFurnace || (game.shelter.built && game.shelter.level >= 0);
            let clothesProtection = game.equippedClothes ? game.equippedClothes.warmth : 0.0;
            const tempLabel = document.getElementById('temp-status');
            const warningDiv = document.getElementById('freeze-warning');

            if (isNight && !hasWarmth) {
                tempLabel.innerText = "極寒 (-15°C)"; tempLabel.className = "font-bold text-sky-400 animate-pulse";
                warningDiv.classList.remove('hidden');
                let rawDamage = exactFreezeDamage * (1.0 - clothesProtection);
                let finalDmg = Math.max(1, Math.round(rawDamage));
                game.hp = Math.max(0, game.hp - finalDmg);
                playSound('freeze');
                exactFreezeDamage *= 1.1;
                document.getElementById('freeze-dmg-show').innerText = Math.round(exactFreezeDamage);
            } else {
                tempLabel.innerText = isNight ? "涼爽 (10°C)" : "舒適 (24°C)";
                tempLabel.className = "font-bold text-emerald-400";
                warningDiv.classList.add('hidden');
                exactFreezeDamage = 2.0;
            }

            if (game.hp <= 0) {
                isGameRunning = false;
                showModal("💀 你死去了", `你未能捱過末日考驗。存活了 ${game.day} 天。`);
                startGame(false);
            }
            updateUI();
        }

        function performAction() {
            if (!isGameRunning) return false;
            
            if (game.equippedWeapon !== 'none') {
                fireBullet();
                return true;
            }

            let nearestObj = null; let minDist = 75;
            mapObjects.forEach(obj => {
                let d = Math.hypot(game.player.x - obj.x, game.player.y - obj.y);
                if (d < minDist) { nearestObj = obj; minDist = d; }
            });

            if (nearestObj) {
                let currentTotal = getTotalItemsCount();
                let backpackCap = getBackpackCapacity();
                if (currentTotal >= backpackCap && (nearestObj.type === 'tree' || nearestObj.type === 'steel_pile' || nearestObj.type === 'leather_pile' || nearestObj.type === 'animal')) {
                    showModal("🎒 背包已滿！", `你目前的物品數為 ${currentTotal}/${backpackCap}，無法繼續收集資源，請升級背包或消耗物資。`);
                    return false;
                }

                if (nearestObj.type === 'tree') {
                    playSound('chop');
                    let power = game.equippedAxe ? game.equippedAxe.efficiency : 1;
                    nearestObj.hp -= power;
                    let got = Math.floor(2 + Math.random() * 3) * power;
                    if (currentTotal + got > backpackCap) got = Math.max(0, backpackCap - currentTotal);
                    game.resources.wood += got; logMsg(`🪓 砍樹，獲得木頭 x${got}`, "success");
                    if (nearestObj.hp <= 0) {
                        mapObjects = mapObjects.filter(o => o.id !== nearestObj.id);
                        setTimeout(() => { mapObjects.push(createMapObject('tree', Math.random()*WORLD_SIZE, Math.random()*WORLD_SIZE)); }, 3000);
                    }
                    updateUI(); return true;
                }
                else if (nearestObj.type === 'steel_pile') {
                    playSound('hit');
                    nearestObj.hp--;
                    let got = Math.floor(1 + Math.random() * 2);
                    if (currentTotal + got > backpackCap) got = Math.max(0, backpackCap - currentTotal);
                    game.resources.steel += got; logMsg(`⛏️ 採集廢鋼堆，獲得鋼材 x${got}`, "success");
                    if (nearestObj.hp <= 0) { mapObjects = mapObjects.filter(o => o.id !== nearestObj.id); }
                    updateUI(); return true;
                }
                else if (nearestObj.type === 'leather_pile') {
                    playSound('hit');
                    nearestObj.hp--;
                    let got = Math.floor(1 + Math.random() * 3);
                    if (currentTotal + got > backpackCap) got = Math.max(0, backpackCap - currentTotal);
                    game.resources.leather += got; logMsg(`🧶 搜刮皮卷，獲得皮革 x${got}`, "success");
                    if (nearestObj.hp <= 0) { mapObjects = mapObjects.filter(o => o.id !== nearestObj.id); }
                    updateUI(); return true;
                }
                else if (nearestObj.type === 'animal') {
                    playSound('hit');
                    nearestObj.hp--;
                    if (nearestObj.hp <= 0) {
                        let gotMeat = Math.floor(2 + Math.random() * 3);
                        let gotLeather = Math.floor(1 + Math.random() * 2);
                        
                        if (currentTotal + gotMeat > backpackCap) gotMeat = Math.max(0, backpackCap - currentTotal);
                        game.resources.rawMeat += gotMeat;
                        currentTotal = getTotalItemsCount();
                        
                        if (currentTotal + gotLeather > backpackCap) gotLeather = Math.max(0, backpackCap - currentTotal);
                        game.resources.leather += gotLeather;

                        logMsg(`⚔️ 擊殺獵物，取得生肉 x${gotMeat}，皮革 x${gotLeather}！`, "success");
                        mapObjects = mapObjects.filter(o => o.id !== nearestObj.id);
                        setTimeout(() => { mapObjects.push(createMapObject('animal', Math.random()*WORLD_SIZE, Math.random()*WORLD_SIZE)); }, 5000);
                    } else {
                        logMsg("⚔️ 打擊獵物！獵物驚慌四散。", "info");
                        nearestObj.animalWalkDir = Math.atan2(nearestObj.y - game.player.y, nearestObj.x - game.player.x);
                        nearestObj.animalWalkTimer = 0;
                    }
                    updateUI(); return true;
                }
                else if (nearestObj.type === 'bed') {
                    logMsg("💤 躺在床上睡覺...", "info");
                    game.stamina = 500;
                    game.hp = Math.min(100, game.hp + 20);
                    game.timeOfDay = 0; 
                    game.day++;
                    logMsg(`☀️ 睡醒了！新的一天。天數: ${game.day}`, "success");
                    saveGame();
                    showModal("💤 睡覺成功", `你安穩地度過了黑夜！進度已儲存。`); playSound('chime');
                    updateUI(); return true;
                }
                else if (nearestObj.type === 'furnace') {
                    if (game.resources.rawMeat > 0) {
                        game.resources.rawMeat--; game.resources.cookedMeat++;
                        logMsg("🔥 熔爐烤熟生肉，獲得熟肉 x1", "success"); playSound('craft');
                    } else { logMsg("沒有生肉可以烹飪。", "info"); }
                    updateUI(); return true;
                }
            }
            return false; 
        }

        function upgradeWorkbench() {
            if (game.workbenchLvl === 1) {
                if (game.resources.wood >= 50 && game.resources.steel >= 30) {
                    game.resources.wood -= 50; game.resources.steel -= 30; game.workbenchLvl++;
                    playSound('craft'); logMsg(`🔧 工作台升級到 LV.${game.workbenchLvl}！`, "success"); 
                    updateWorkbenchOverlayUI(); updateUI();
                } else { logMsg("升級不足！需木頭x50, 鋼鐵x30", "warning"); }
            } else if (game.workbenchLvl === 2) {
                if (game.resources.wood >= 100 && game.resources.steel >= 80) {
                    game.resources.wood -= 100; game.resources.steel -= 80; game.workbenchLvl++;
                    playSound('craft'); logMsg(`🔧 工作台升級到 LV.${game.workbenchLvl}！`, "success"); 
                    updateWorkbenchOverlayUI(); updateUI();
                } else { logMsg("升級不足！需木頭x100, 鋼鐵x80", "warning"); }
            } else {
                logMsg("工作台已達到最高等級 (LV.3)！", "info");
            }
        }

        function handleShelterAction() {
            let nextLvl = game.shelter.level + 1;
            let costWood = 30 * (nextLvl > 0 ? Math.pow(2, nextLvl) : 1);
            let costSteel = 20 * (nextLvl > 0 ? Math.pow(2, nextLvl) : 1);

            if (nextLvl === 0) { costWood = 30; costSteel = 20; }

            if (game.resources.wood < costWood || game.resources.steel < costSteel) {
                logMsg(`資源不足以建造/升級庇護所！`, "danger"); return;
            }

            game.resources.wood -= costWood;
            game.resources.steel -= costSteel;

            let daysToTake = nextLvl === 1 ? 1 : (nextLvl === 2 ? 3 : (nextLvl === 3 ? 5 : (nextLvl === 4 ? 10 : 1)));
            game.shelter.upgrading = true; game.shelter.daysRemaining = daysToTake; game.shelter.targetLevel = nextLvl;

            if (nextLvl === 0) {
                let newShelter = createMapObject('shelter', game.player.x + 40, game.player.y - 40, 0);
                newShelter.level = 0; mapObjects.push(newShelter);
            }
            logMsg(`🔨 施工建造/升級！預計 ${daysToTake} 天。`, "success"); playSound('craft'); updateUI();
        }

        function equipWeapon(type) {
            game.equippedWeapon = type;
            logMsg(`🔫 當前持槍裝備切換為：${type === 'none' ? '採集工具/斧頭' : type === 'm4a1' ? 'M4A1 突擊步槍' : '.357 左輪手槍'}`);
            updateUI();
        }

        function craftItem(recipeId) {
            const recipe = RECIPES.find(r => r.id === recipeId);
            let nearWorkbench = false;
            mapObjects.forEach(obj => {
                if (obj.type === 'workbench' && Math.hypot(game.player.x - obj.x, game.player.y - obj.y) < 70) nearWorkbench = true;
            });

            if (!nearWorkbench) { showModal("遠離工作台", "請靠近工作台旁才能合成。"); return; }
            if (game.workbenchLvl < recipe.minLvl) { logMsg(`工作台等級不足，需要 LV.${recipe.minLvl}`, "warning"); return; }

            let cap = getBackpackCapacity();
            let current = getTotalItemsCount();
            
            let totalCostItems = 0;
            for (let r in recipe.cost) { totalCostItems += recipe.cost[r]; }
            let netGainItems = 0;
            if (recipeId === 'm4a1_ammo') netGainItems = 30;
            else if (recipeId === 'revolver_ammo') netGainItems = 10;
            
            if (current - totalCostItems + netGainItems > cap) {
                showModal("🎒 背包空間不足！", `合成此物品將使得物資超出背包上限。現有容量：${current}/${cap}`);
                return;
            }

            for (let res in recipe.cost) {
                if (game.resources[res] < recipe.cost[res]) {
                    logMsg(`合成材料不足！`, "danger"); return;
                }
            }
            for (let res in recipe.cost) { game.resources[res] -= recipe.cost[res]; }

            if (recipeId === 'stone_axe') { game.equippedAxe = { name: "石製斧頭", quality: 1, efficiency: 2 }; }
            else if (recipeId === 'steel_axe') { game.equippedAxe = { name: "精鋼手斧", quality: 2, efficiency: 3 }; }
            else if (recipeId === 'clothes_1') { game.equippedClothes = { name: "一般毛衣", warmth: 0.25 }; }
            else if (recipeId === 'clothes_2') { game.equippedClothes = { name: "防寒軍大衣", warmth: 0.60 }; }
            else if (recipeId === 'bed_1') { game.equippedBed = { name: "暖木吊床", quality: 1, staminaRate: 10 }; }
